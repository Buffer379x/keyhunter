// Cover art: a stand-in for IGDB covers (no internet in the audit). Gradient, a shape and the title, 264 x 352 like t_cover_big.
const palette = [[18, 40], [210, 230], [28, 8], [265, 300], [140, 170], [0, 340], [190, 220], [45, 25], [300, 330], [100, 130]];
export function coverSvg(title: string, i: number) {
  const [a, b] = palette[i % palette.length];
  const words = title.replace(/[:–-].*$/, '').split(' ');
  const lines = words.reduce<string[]>((acc, w) => { const l = acc[acc.length - 1]; if (l && (l + ' ' + w).length < 12) acc[acc.length - 1] = l + ' ' + w; else acc.push(w); return acc; }, []).slice(0, 3);
  return `<svg xmlns="http://www.w3.org/2000/svg" width="264" height="352" viewBox="0 0 264 352"><defs><linearGradient id="g" x1="0" y1="0" x2="1" y2="1"><stop offset="0" stop-color="hsl(${a} 55% 42%)"/><stop offset="1" stop-color="hsl(${b} 45% 12%)"/></linearGradient></defs>
<rect width="264" height="352" fill="url(#g)"/><circle cx="${60 + (i * 37) % 150}" cy="${90 + (i * 53) % 120}" r="${70 + (i * 13) % 50}" fill="hsl(${a} 70% 70% / .18)"/>
<path d="M0 ${260 - i * 4} L264 ${200 + i * 5} L264 352 L0 352Z" fill="rgb(0 0 0 / .35)"/>
${lines.map((l, k) => `<text x="20" y="${270 + k * 30}" font-family="Arial, sans-serif" font-weight="700" font-size="26" fill="white">${l.toUpperCase()}</text>`).join('')}</svg>`;
}

/** Mocks every list endpoint; returns the id of the real (seeded) Forza wish. */
export async function mockAll(page: Page) {
  let forzaId = '';
  await page.route('https://images.igdb.com/**', r => {
    const url = new URL(r.request().url());
    const m = url.pathname.match(/audit(\d+)/);
    const i = Math.abs(m ? Number(m[1]) - 5000 : 0);
    const title = url.searchParams.get('t') ?? 'Forza Horizon 5';
    return r.fulfill({ contentType: 'image/svg+xml', body: coverSvg(title, i) });
  });
  await page.route('**/api/v1/wishlist', async r => {
    if (r.request().method() !== 'GET') return r.continue();
    const rows = await (await r.fetch()).json();
    forzaId = rows[0]?.id ?? '';
    const real = rows.map((x: Record<string, unknown>) => ({ ...x, game: x.game ? { ...(x.game as object), cover_url: 'https://images.igdb.com/igdb/image/upload/t_cover_big/audit4990.jpg?t=Forza%20Horizon%205' } : x.game, price_badge: forzaBadge, offer_count: 5 }));
    await r.fulfill({ json: [...real, ...syntheticRows] });
  });
  await page.route(/\/api\/v1\/wishlist\/price-badges(\?.*)?$/, r => r.fulfill({ json: Object.fromEntries([[forzaId, forzaBadge], ...syntheticRows.map(x => [x.id, x.price_badge])]) }));
  await page.route('**/api/v1/wishlist/*/offers', r => r.request().method() === 'GET' ? r.fulfill({ json: offers }) : r.continue());
  await page.route('**/api/v1/wishlist/*/alert', r => r.request().method() === 'GET' ? r.fulfill({ json: { rule: { mode: 'target', basis: 'base', target_cents: 2900, enabled: true }, configured: true, channels: ['Telegram'] } }) : r.continue());
  await page.route('**/api/v1/alerts', r => r.request().method() === 'GET' ? r.fulfill({ json: {
    configured: true, channels: ['Telegram', 'E-Mail'],
    rules: [
      { wish_id: forzaId, title: 'Forza Horizon 5', mode: 'target', basis: 'base', target_cents: 2900, enabled: true, delivery: null },

    ],
  } }) : r.continue());
  const offer = (cents: number, region: string, stale = false) => ({ cents, estimate_cents: Math.round(cents * 1.04), region, observed_at: now - 2 * H, stale });
  await page.route('**/api/v1/cart', async r => {
    if (r.request().method() !== 'GET') return r.continue();
    const real = await (await page.request.get('/api/v1/wishlist')).json();
    const forza = real[0];
    const entries = [
      { ...forza, price_badge: undefined, added_at: now - D, source: null, scope: 'world', offers: { eneba: { home: offer(2890, 'Global'), world: offer(1290, 'India') }, driffle: { home: offer(3000, 'DE'), world: offer(1490, 'Argentina') } } },

}
